import 'package:flutter/widgets.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:morph/motion.dart';
import 'package:morph_example/proto/menu_route_probe.dart';

const _button = Rect.fromLTWH(24, 80, 44, 44);
const _bounds = Size(402, 874);

class _Harness {
  final source = GlobalKey<ProbeMenuSourceState>();
  late MorphMenuMotion model;

  Widget app(WidgetTester tester, {bool scaledRoot = false}) {
    final app = WidgetsApp(
      color: const Color(0xFFFFFFFF),
      onGenerateRoute: (_) => PageRouteBuilder<void>(
        transitionDuration: Duration.zero,
        pageBuilder: (_, _, _) => Padding(
          padding: const EdgeInsets.only(left: 80, top: 40),
          child: ProbeMenuSource(
            key: source,
            pointerInput: true,
            now: () => Duration(
              microseconds: tester.binding.clock.now().microsecondsSinceEpoch,
            ),
            createMotion: () => model = MorphMenuMotion(
              button: _button,
              bounds: _bounds,
              itemCount: 3,
              sourceHeight: 64,
              sourceRadius: 22,
            ),
            menuBuilder: (_) => const SizedBox.expand(),
          ),
        ),
      ),
    );
    return scaledRoot
        ? Transform.scale(scale: 0.8, alignment: .topLeft, child: app)
        : app;
  }

  Finder get button => find.descendant(
    of: find.byKey(source),
    matching: find.byWidgetPredicate(
      (widget) =>
          widget is Semantics &&
          widget.properties.label == 'Open prototype menu',
    ),
  );

  Future<void> mount(WidgetTester tester, {bool scaledRoot = false}) async {
    tester.view.devicePixelRatio = 1;
    tester.view.physicalSize = _bounds;
    addTearDown(tester.view.reset);
    await tester.pumpWidget(app(tester, scaledRoot: scaledRoot));
  }
}

void main() {
  for (final scaledRoot in [false, true]) {
    testWidgets(
      'real source hit opens after offset, scaled root: $scaledRoot',
      (tester) async {
        final h = _Harness();
        await h.mount(tester, scaledRoot: scaledRoot);
        await tester.tapAt(tester.getCenter(h.button));
        await tester.pump(const Duration(milliseconds: 300));
        expect(h.model.isOpen, isTrue);
      },
    );
  }

  testWidgets('first route frame starts at the measured source rectangle', (
    tester,
  ) async {
    final h = _Harness();
    await h.mount(tester);
    final physicalButton = tester.getRect(h.button);
    final route = h.source.currentState!.open();
    expect(route.frame.button.outerRect.center, physicalButton.center);
  });
}
